The host owns the connection and credentials; an unreachable server is a ConfigError naming it.

Import from `@threads/mcp` (TypeScript) or `threads.mcp` (Python).

<Tabs items={["TypeScript", "Python"]} groupId="lang" persist>
<Tab value="TypeScript">

```ts
function mcp(options: {
  name: Name;
  url?: string;
  headers?: Record<string, string | Secret>;
  command?: string;
  args?: readonly string[];
  env?: Record<string, string | Secret>;
  runs?: "host" | "sandbox";
  tools?: { allow?: readonly string[]; deny?: readonly string[] };
  effect?: EffectClass;
}): McpServer
```

## Parameters

<Field name="name" type={"Name"} required>
  Tools are named mcp__&lt;name>__&lt;tool>.
</Field>

<Field name="url" type={"string"}>
  Streamable HTTP (SSE fallback). Exactly one of url or command.
</Field>

<Field name="headers" type={"Record<string, string | Secret>"} />

<Field name="command" type={"string"}>
  stdio transport.
</Field>

<Field name="args" type={"readonly string[]"} />

<Field name="env" type={"Record<string, string | Secret>"} />

<Field name="runs" type={"\"host\" | \"sandbox\""} default={"host"}>
  sandbox: stdio server inside the sandbox, with no host credentials. Only "host" is supported today; "sandbox" is a setup error.
</Field>

<Field name="tools" type={"{ allow?: readonly string[]; deny?: readonly string[] }"}>
  Filter before pinning.
</Field>

<Field name="effect" type={"EffectClass"} default={"unguarded"} />
